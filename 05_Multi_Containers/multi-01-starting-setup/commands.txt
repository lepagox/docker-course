# Backend

# build image then run
docker build -t goals-node .

docker run --name goals-backend --rm -d -p 80:80 goals-node



# Frontend

# build image then run
docker build -t goals-react . 
docker run --name goals-frontend --rm -d -it -p 3000:3000 goals-react


# network
docker network create goals

# restart backen container to use network - frontend doesn't need network as 
# React code goes from browser to backend node, then run db container
docker run --name goals-backend --rm -d --network goals -p 80:80 goals-node
docker run --name goals-frontend --rm -d -p 3000:3000 -it goals-react
docker run -d --rm --name mongodb --network goals mongo

# make data persistent
# stop mongo then restart with command below - now data should be kept between restarts
docker run --name mongodb -v data:/data/db -d --rm --network goals mongo

# run backend again but with live code (bind mount)
docker run --name goals-backend -v "$(pwd):/app" -v logs:/app/logs -v /app/node_modules --rm -d --network goals -p 80:80 goals-node

# now to run db with creds
docker run --name mongodb -v data:/data/db -d --rm --network goals -e MONGO_INITDB_ROOT_USERNAME=peng -e MONGO_INITDB_ROOT_PASSWORD=mima mongo

# after removing hard-coded mongo creds (put in env file or Dockerfile) restart with overriding creds in command line
# see comment in app.js mongoose.connect()
docker run --name goals-backend -v "$(pwd):/app" -v logs:/app/logs -v /app/node_modules -e MONGODB_USERNAME=peng --rm  --network goals -p 80:80 goals-node

# update React to have a Bind Mount to live code
docker run -v "$(pwd)/src:/app/src" --name goals-frontend --rm -d -p 3000:3000 --network goals -it goals-react
